cmake_minimum_required(VERSION 3.16)
project(BioresonanceGUI VERSION 1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(BIORESONANCE_WARNINGS_AS_ERRORS "Treat warnings as errors" OFF)
find_package(SDL2 CONFIG REQUIRED)
find_package(Threads REQUIRED)
set(IMGUI_SOURCES
 imgui_lib/imgui.cpp imgui_lib/imgui_demo.cpp imgui_lib/imgui_draw.cpp
 imgui_lib/imgui_tables.cpp imgui_lib/imgui_widgets.cpp
 imgui_lib/backends/imgui_impl_sdl2.cpp
 imgui_lib/backends/imgui_impl_sdlrenderer2.cpp
 imgui_lib/misc/cpp/imgui_stdlib.cpp)
set(SOURCES src/main.cpp src/gui_app.cpp src/network_client.cpp
 src/probe_manager.cpp src/frequency_loader.cpp ${IMGUI_SOURCES})
set(HEADERS include/gui_app.h include/network_client.h include/probe_manager.h
 include/frequency_loader.h include/types.h include/logger.h
 include/event_system.h include/error_handler.h)
add_executable(bioresonance_gui ${SOURCES} ${HEADERS})
target_include_directories(bioresonance_gui PRIVATE include imgui_lib
 imgui_lib/backends imgui_lib/misc/cpp)
if(TARGET SDL2::SDL2)
 target_link_libraries(bioresonance_gui PRIVATE SDL2::SDL2)
elseif(TARGET SDL2::SDL2-static)
 target_link_libraries(bioresonance_gui PRIVATE SDL2::SDL2-static)
else()
 message(FATAL_ERROR "SDL2 found without a supported CMake target")
endif()
if(WIN32)
 target_compile_definitions(bioresonance_gui PRIVATE _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN NOMINMAX)
 target_link_libraries(bioresonance_gui PRIVATE ws2_32 imm32 winmm)
elseif(APPLE)
 find_library(COCOA_FRAMEWORK Cocoa REQUIRED)
 find_library(METAL_FRAMEWORK Metal REQUIRED)
 find_library(METALKIT_FRAMEWORK MetalKit REQUIRED)
 target_link_libraries(bioresonance_gui PRIVATE ${COCOA_FRAMEWORK} ${METAL_FRAMEWORK} ${METALKIT_FRAMEWORK})
elseif(UNIX)
 target_link_libraries(bioresonance_gui PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
endif()
if(MSVC)
 target_compile_options(bioresonance_gui PRIVATE /W4)
 if(BIORESONANCE_WARNINGS_AS_ERRORS)
  target_compile_options(bioresonance_gui PRIVATE /WX)
 endif()
else()
 target_compile_options(bioresonance_gui PRIVATE -Wall -Wextra -Wpedantic)
 if(BIORESONANCE_WARNINGS_AS_ERRORS)
  target_compile_options(bioresonance_gui PRIVATE -Werror)
 endif()
endif()
configure_file(${CMAKE_CURRENT_SOURCE_DIR}/../frequencies.md
 ${CMAKE_CURRENT_BINARY_DIR}/frequencies.md COPYONLY)
install(TARGETS bioresonance_gui RUNTIME DESTINATION bin)

include(CTest)
if(BUILD_TESTING)
 add_executable(logger_smoke tests/logger_smoke.cpp)
 target_include_directories(logger_smoke PRIVATE include)
 target_link_libraries(logger_smoke PRIVATE Threads::Threads)
 add_test(NAME logger_reentrancy COMMAND logger_smoke)

 add_executable(error_handler_smoke tests/error_handler_smoke.cpp)
 target_include_directories(error_handler_smoke PRIVATE include)
 target_link_libraries(error_handler_smoke PRIVATE Threads::Threads)
 add_test(NAME error_handler_reentrancy COMMAND error_handler_smoke)
 set_tests_properties(error_handler_reentrancy PROPERTIES TIMEOUT 10)

 add_executable(frequency_loader_smoke tests/frequency_loader_smoke.cpp src/frequency_loader.cpp)
 target_include_directories(frequency_loader_smoke PRIVATE include)
 target_link_libraries(frequency_loader_smoke PRIVATE Threads::Threads)
 add_test(NAME frequency_loader_regression COMMAND frequency_loader_smoke)

 add_executable(network_client_smoke tests/network_client_smoke.cpp src/network_client.cpp)
 target_include_directories(network_client_smoke PRIVATE include)
 target_link_libraries(network_client_smoke PRIVATE Threads::Threads)
 if(WIN32)
  target_compile_definitions(network_client_smoke PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(network_client_smoke PRIVATE ws2_32)
 endif()
 add_test(NAME network_client_loopback COMMAND network_client_smoke)
 set_tests_properties(network_client_loopback PROPERTIES TIMEOUT 15)
endif()

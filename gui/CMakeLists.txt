cmake_minimum_required(VERSION 3.21)
project(BioresonanceDesktop VERSION 2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Charts)
find_package(Threads REQUIRED)
add_executable(bioresonance_gui qt/main.cpp)
target_link_libraries(bioresonance_gui PRIVATE Qt6::Widgets Qt6::Charts)
target_compile_options(bioresonance_gui PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic> $<$<CXX_COMPILER_ID:MSVC>:/W4>)
include(CTest)
if(BUILD_TESTING)
 add_executable(resonet_protocol_v1_smoke tests/resonet_protocol_v1_smoke.cpp)
 target_include_directories(resonet_protocol_v1_smoke PRIVATE include)
 add_test(NAME resonet_protocol_v1_codec COMMAND resonet_protocol_v1_smoke)
 add_executable(logger_smoke tests/logger_smoke.cpp)
 target_include_directories(logger_smoke PRIVATE include)
 target_link_libraries(logger_smoke PRIVATE Threads::Threads)
 add_test(NAME logger_reentrancy COMMAND logger_smoke)
 add_executable(error_handler_smoke tests/error_handler_smoke.cpp)
 target_include_directories(error_handler_smoke PRIVATE include)
 target_link_libraries(error_handler_smoke PRIVATE Threads::Threads)
 add_test(NAME error_handler_reentrancy COMMAND error_handler_smoke)
 set_tests_properties(error_handler_reentrancy PROPERTIES TIMEOUT 10)
 add_executable(frequency_loader_smoke tests/frequency_loader_smoke.cpp src/frequency_loader.cpp)
 target_include_directories(frequency_loader_smoke PRIVATE include)
 target_link_libraries(frequency_loader_smoke PRIVATE Threads::Threads)
 add_test(NAME frequency_loader_regression COMMAND frequency_loader_smoke)
 add_executable(network_client_smoke tests/network_client_smoke.cpp src/network_client.cpp)
 target_include_directories(network_client_smoke PRIVATE include)
 target_link_libraries(network_client_smoke PRIVATE Threads::Threads)
 if(WIN32)
  target_compile_definitions(network_client_smoke PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(network_client_smoke PRIVATE ws2_32)
 endif()
 add_test(NAME network_client_loopback COMMAND network_client_smoke)
 set_tests_properties(network_client_loopback PROPERTIES TIMEOUT 15)
endif()
install(TARGETS bioresonance_gui RUNTIME DESTINATION bin)
